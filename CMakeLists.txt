cmake_minimum_required(VERSION 3.24)
project(VisionFlow VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(VISIONFLOW_BUILD_UI "Build Qt Widgets desktop and UI tests" ON)
include(CTest)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Concurrent)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
add_library(visionflow_matching src/vision/matching/TemplateMatcher.cpp)
target_include_directories(visionflow_matching PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(visionflow_matching PUBLIC ${OpenCV_LIBS})
add_executable(match_benchmark benchmarks/match_benchmark.cpp)
target_link_libraries(match_benchmark PRIVATE visionflow_matching)
add_library(visionflow_projects src/domain/Project.cpp src/domain/Coordinates.cpp src/domain/ToolDefinition.cpp src/domain/ToolGraph.cpp src/domain/ToolRegistry.cpp src/persistence/ProjectRepository.cpp src/application/ProjectService.cpp)
target_include_directories(visionflow_projects PUBLIC src)
target_link_libraries(visionflow_projects PUBLIC Qt6::Core)
add_library(visionflow_hardware src/hardware/GenTLTransport.cpp)
target_include_directories(visionflow_hardware PUBLIC src PRIVATE third_party/gentl)
target_link_libraries(visionflow_hardware PUBLIC Qt6::Core)
add_library(visionflow_acquisition src/vision/ImageFrame.cpp src/application/ImageFileService.cpp src/application/AcquisitionService.cpp src/application/CameraService.cpp src/application/CameraLifecycle.cpp src/application/MatchService.cpp src/application/ExecutionService.cpp src/application/ToolExecutors.cpp src/application/EditorSession.cpp)
target_include_directories(visionflow_acquisition PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_sources(visionflow_acquisition PRIVATE src/application/ProductionRecipeService.cpp)
target_link_libraries(visionflow_acquisition PUBLIC Qt6::Gui Qt6::Concurrent visionflow_projects visionflow_matching visionflow_hardware ${OpenCV_LIBS})
add_executable(visionflow_camera_probe src/app/camera_probe.cpp)
target_link_libraries(visionflow_camera_probe PRIVATE visionflow_acquisition)
if(VISIONFLOW_BUILD_UI)
  find_package(Qt6 REQUIRED COMPONENTS Widgets)
  add_library(visionflow_viewer src/app/ImageViewer.cpp)
  target_include_directories(visionflow_viewer PUBLIC src)
  target_link_libraries(visionflow_viewer PUBLIC Qt6::Widgets)
  qt_add_executable(visionflow src/app/main.cpp src/app/MainWindow.cpp src/app/MainWindowWorkspace.cpp src/app/MainWindowEditor.cpp src/app/MainWindowMatching.cpp src/app/MainWindowExecution.cpp src/app/MainWindowCamera.cpp src/app/EditorSmoke.cpp)
  qt_add_resources(visionflow "theme" PREFIX "/" FILES resources/theme.qss)
  target_link_libraries(visionflow PRIVATE Qt6::Widgets Qt6::Concurrent visionflow_projects visionflow_acquisition visionflow_viewer)
endif()

  target_link_libraries(execution_test PRIVATE visionflow_acquisition)
  add_test(NAME tool_execution COMMAND execution_test)
  add_executable(acquisition_test tests/acquisition_test.cpp)
  target_link_libraries(acquisition_test PRIVATE visionflow_acquisition)
  add_test(NAME acquisition_contract COMMAND acquisition_test)
  add_executable(session_test tests/session_test.cpp)
  target_link_libraries(session_test PRIVATE visionflow_acquisition)
  add_test(NAME session_contract COMMAND session_test)
  add_executable(frame_test tests/frame_test.cpp)

  add_executable(coordinates_test tests/coordinates_test.cpp)

  add_test(NAME coordinates_contract COMMAND coordinates_test)
  add_executable(production_recipe_test tests/production_recipe_test.cpp)

  target_link_libraries(camera_lifecycle_test PRIVATE visionflow_acquisition)

  add_library(gentl_test_producer SHARED tests/gentl_test_producer.cpp)
  target_include_directories(gentl_test_producer PRIVATE third_party/gentl)
  set_target_properties(gentl_test_producer PROPERTIES PREFIX "" SUFFIX ".cti")

  set_target_properties(gentl_incomplete_producer PROPERTIES PREFIX "" SUFFIX ".cti")
