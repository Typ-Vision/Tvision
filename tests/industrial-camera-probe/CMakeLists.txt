cmake_minimum_required(VERSION 3.24)
project(VisionFlowIndustrialBackendProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This opt-in experiment requires Windows x64")
endif()
set(OECS_INCLUDE_DIR "" CACHE PATH "Local OECS Development/Includes (not redistributed)")
find_path(OECS_HEADER MvCameraControl.h PATHS "${OECS_INCLUDE_DIR}" NO_DEFAULT_PATH REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(industrial_backend_probe main.cpp)
target_include_directories(industrial_backend_probe PRIVATE "${OECS_HEADER}")
target_link_libraries(industrial_backend_probe PRIVATE Qt6::Core)
add_executable(oecs_trigger_diagnostic trigger.cpp)
target_include_directories(oecs_trigger_diagnostic PRIVATE "${OECS_HEADER}")
target_link_libraries(oecs_trigger_diagnostic PRIVATE Qt6::Core)
