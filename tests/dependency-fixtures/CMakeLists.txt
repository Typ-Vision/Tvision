cmake_minimum_required(VERSION 3.24)
project(VisionFlowIndependentCodeFixtures LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT EXISTS "${VISIONFLOW_ZINT_SOURCE}/backend/zint.h")
  message(FATAL_ERROR "Run scripts/bootstrap-zxing.ps1 first; local Zint source required")
endif()
# Only libzint, no frontend, PNG, Qt or decoder dependency.
set(ZINT_SHARED OFF CACHE BOOL "" FORCE)
set(ZINT_STATIC ON CACHE BOOL "" FORCE)
set(ZINT_USE_PNG OFF CACHE BOOL "" FORCE)
set(ZINT_USE_QT OFF CACHE BOOL "" FORCE)
set(ZINT_FRONTEND OFF CACHE BOOL "" FORCE)
set(ZINT_TEST OFF CACHE BOOL "" FORCE)
set(ZINT_UNINSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${VISIONFLOW_ZINT_SOURCE}" zint EXCLUDE_FROM_ALL)
add_executable(code_fixture_generator code_fixture_generator.cpp)
target_include_directories(code_fixture_generator PRIVATE "${VISIONFLOW_ZINT_SOURCE}/backend")
target_link_libraries(code_fixture_generator PRIVATE zint-static)
